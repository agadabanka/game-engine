\section{Overview}
This document is typeset by the engine's standard \textbf{TeX backend}
(Tectonic). Inline math renders cleanly --- e.g.\ $\nabla\cdot\mathbf{E}=\rho/\varepsilon_0$.

\callout{All new and rewritten PDFs use this backend. The Chromium / Gemini-art
path (the \texttt{book} skill) is reserved for image-heavy illustrated books.}

\section{Structure}
\begin{itemize}[noitemsep]
  \item portable preamble --- compiles with no system fonts
  \item headings, footer, code, tables, math, and links, styled consistently
\end{itemize}
